;libreria base
.include "m328pdef.inc"

.cseg
.org 0x00
rjmp inicio

;componente para inicializacion comunicacion UART
.equ f_cpu = 16000000
.equ baud = 9600
.equ bps = (f_cpu/16/baud) - 1


inicio:
	;inicializacion stack pointer
	ldi r16, HIGH(RAMEND)
	out SPH, r16
	ldi r16, LOW(RAMEND)
	out SPL, r16
	
	;inicializacion comunicacion UART
	ldi r16, LOW(bps)
	ldi r17, HIGH(bps)
	rcall initUART

	;se inicializa puertos B0-B2 como entradas en r19
	ldi r19,0x00 
	out DDRB, r19; 
	ldi r19, 0x07 
	out  PORTB, r19

main_loop:

	;reiniciamos el valor de r16
	ldi r16, 0

	;actualizamos r16 dependiendo boton apretado con sumas (or's)
	sbis PINB, PB0
	ori  r16, 0b00000001

	sbis PINB, PB1
	ori  r16, 0b00000010

	sbis PINB, PB2
	ori  r16, 0b00000100

	;llamada a funcion para enviar r16
	rcall putnum

	rjmp main_loop

;activacion de puertos rx y tx para comunicacion UART
initUART:
	sts UBRR0L,r16
	sts UBRR0H,r17

	ldi r16,(1<<RXEN0)|(1<<TXEN0)
	sts UCSR0B,r16
	ret

;enviar valor en r16 (botones apretados) por UART
putnum:
	lds r17, UCSR0A
	sbrs r17, UDRE0
	rjmp putnum
	sts UDR0, r16
	ldi r16, 0
	ret
