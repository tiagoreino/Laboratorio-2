.include "m328pdef.inc"

.cseg
.org 0x00
rjmp inicio

.equ f_cpu = 16000000
.equ baud = 9600
.equ bps = (f_cpu/16/baud) - 1

inicio:
	ldi r16, HIGH(RAMEND)
	out SPH, r16
	ldi r16, LOW(RAMEND)
	out SPL, r16
	
	ldi r16, LOW(bps)
	ldi r17, HIGH(bps)
	rcall initUART

	ldi r19,0x00 
	out DDRB, r19; 
	ldi r19, 0x07 
	out  PORTB, r19

main_loop:
	ldi r16, 0

	sbis PINB, PB0
	ori  r16, 0b00000001

	sbis PINB, PB1
	ori  r16, 0b00000010

	sbis PINB, PB2
	ori  r16, 0b00000100

	rcall putnum

	rjmp main_loop


initUART:
	sts UBRR0L,r16
	sts UBRR0H,r17

	ldi r16,(1<<RXEN0)|(1<<TXEN0)
	sts UCSR0B,r16
	ret


putnum:
	lds r17, UCSR0A
	sbrs r17, UDRE0
	rjmp putnum
	sts UDR0, r16
	ldi r16, 0
	ret
