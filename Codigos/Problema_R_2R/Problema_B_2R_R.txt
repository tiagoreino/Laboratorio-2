.include "m328pdef.inc"

.ORG 0x0000
RJMP start

.org OC1Aaddr
    rjmp TIMER1_COMPA
    ; Cuando ocurre la interrupción salta a TIMER1_COMPA


.equ F_CPU = 16000000
.equ baud  = 9600
.equ bps   = (F_CPU/16/baud) - 1

;--------------------------------------------------
; LUT SEÑAL 1
;--------------------------------------------------
.equ TABLA_7SEG = 0x0256

;--------------------------------------------------
; LUT SEÑAL 2
;--------------------------------------------------
.equ TABLA_SENAL2 = 0x0256



;==================================================
; INICIO
;==================================================

start:

;--------------------------------------------------
; Se inicializa Stack Pointer
;--------------------------------------------------

ldi r16, HIGH(RAMEND)
out SPH, r16

ldi r16, LOW(RAMEND)
out SPL, r16


;--------------------------------------------------
; Se inicializa comunicación UART
;
; PD0 = RX
; PD1 = TX
;--------------------------------------------------

ldi r16, LOW(bps)
ldi r17, HIGH(bps)

rcall initUART


;--------------------------------------------------
; Se configuran puertos y se guarda la tabla
;--------------------------------------------------

call configurar
call esperar_inicio


;--------------------------------------------------
; r24 será la posición actual dentro de la LUT
;
; 0,1,2,...,254,255,0,...
;--------------------------------------------------

clr r24


;==================================================
; NUEVO:
; MOSTRAR MENÚ USART
;
; Aquí se guarda en r25:
;
; r25 = 1 -> señal 1
; r25 = 2 -> señal 2
;==================================================

rcall MENU


;--------------------------------------------------
; Configuración de botones PD2 y PD3
;
; Pull-Down externo
;
; 0 = no presionado
; 1 = presionado
;--------------------------------------------------

cbi DDRD, PD2
cbi PORTD, PD2

cbi DDRD, PD3
cbi PORTD, PD3


;--------------------------------------------------
; r17:r18 = valor inicial OCR1A
;--------------------------------------------------

clr r17
clr r18

ldi r17, HIGH(15624)
ldi r18, LOW(15624)


;--------------------------------------------------
; Configurar Timer1
;--------------------------------------------------

rcall CONFIG_TIMER1



;==================================================
; PROGRAMA PRINCIPAL
;==================================================

main_loop:

rjmp ESPERAR_PULSACION



;==================================================
; INICIALIZAR UART
;==================================================

initUART:

sts UBRR0L, r16
sts UBRR0H, r17

ldi r16, (1<<RXEN0)|(1<<TXEN0)
sts UCSR0B, r16

ret



;==================================================
; NUEVO:
; MENÚ DE SELECCIÓN DE SEÑAL
;==================================================

MENU:


;--------------------------------------------------
; Salto de línea
;--------------------------------------------------

rcall putc

ldi r16, 10
rcall putc

; Mostrar:

; MENU

;

rcall putc

;--------------------------------------------------
; Mostrar:
;

;--------------------------------------------------

;==================================================
; ESPERAR OPCIÓN
;==================================================

MENU_ESPERAR:

rcall getc


;--------------------------------------------------
; ¿Se recibió '1'?
;--------------------------------------------------

cpi r16, '1'
breq MENU_SENAL1


;--------------------------------------------------
; ¿Se recibió '2'?
;--------------------------------------------------

cpi r16, '2'
breq MENU_SENAL2


;--------------------------------------------------
; Cualquier otro carácter se ignora
;--------------------------------------------------

rjmp MENU_ESPERAR



;==================================================
; SELECCIONAR SEÑAL 1
;==================================================

MENU_SENAL1:


;--------------------------------------------------
; r25 indica señal seleccionada
;--------------------------------------------------

ldi r25, 1


;--------------------------------------------------
; Siempre comenzar la LUT desde muestra 0
;--------------------------------------------------

clr r24


;--------------------------------------------------
; Mostrar salto de línea
;--------------------------------------------------

ldi r16, 13
rcall putc

ldi r16, 10
rcall putc


ret



;==================================================
; SELECCIONAR SEÑAL 2
;==================================================

MENU_SENAL2:


;--------------------------------------------------
; r25 = 2
;--------------------------------------------------

ldi r25, 2


;--------------------------------------------------
; Comenzar desde muestra 0
;--------------------------------------------------

clr r24

;--------------------------------------------------
; Mostrar salto de línea
;--------------------------------------------------

ldi r16, 13
rcall putc

ldi r16, 10
rcall putc


ret



;==================================================
; WAIT
;==================================================

wait:

call getc

cpi r16, 0
breq wait

ldi r18, 1
add r16, r18

rcall putc

ldi r16, 0

rjmp wait



;==================================================
; TRANSMITIR POR UART
;==================================================

putc:

lds r17, UCSR0A

sbrs r17, UDRE0
rjmp putc

sts UDR0, r16

ldi r16, 0

ret



;==================================================
; RECIBIR POR UART
;==================================================

getc:

lds r17, UCSR0A

sbrs r17, RXC0
rjmp getc

lds r16, UDR0

ret



;==================================================
; CONFIGURAR PUERTOS
;==================================================

configurar:


;--------------------------------------------------
; PC0-PC5:
;
; bits 0-5 del DAC
;--------------------------------------------------

ldi r20, 0b00111111
out DDRC, r20

clr r20
out PORTC, r20


;--------------------------------------------------
; PB0 y PB1:
;
; bits 6-7 del DAC
;
; PB5:
; LED L
;--------------------------------------------------

sbi DDRB, DDB0
sbi DDRB, DDB1
sbi DDRB, DDB5


;--------------------------------------------------
; Valores iniciales DAC = 0
;--------------------------------------------------

cbi PORTB, PB0
cbi PORTB, PB1

ret


;==================================================
; ESPERAR INICIO
;==================================================

esperar_inicio:

nop

ret



;==================================================
; BUSCA VALOR EN LUT
;==================================================

set_7seg_u:

mov r0, r1

call get_7seg_code

mov r17, r20

ret



;==================================================
; BUSCAR VALOR EN LUT DE SEÑAL 1
;
; r0 = índice
;
; r20 = LUT[r0]
;==================================================

get_7seg_code:

;--------------------------------------------------
; Guardar LUT de la señal 1
;--------------------------------------------------

call guardar_codigos


ldi r28, LOW(TABLA_7SEG)
ldi r29, HIGH(TABLA_7SEG)

add r28, r0

clr r19
adc r29, r19

ld r20, Y

ret


;==================================================
; BUSCAR VALOR EN LUT DE SEÑAL 2
;
; r0 = índice
;
; r20 = LUT[r0]
;==================================================

get_senal2_code:

;--------------------------------------------------
; Guardar LUT de la señal 2
;--------------------------------------------------

call guardar_codigos_2


ldi r28, LOW(TABLA_7SEG)
ldi r29, HIGH(TABLA_7SEG)

add r28, r0

clr r19
adc r29, r19

ld r20, Y

ret


;==================================================
; MOSTRAR LOS 8 BITS EN EL DAC
;
; PC0-PC5 = bits 0-5
; PB0-PB1 = bits 6-7
;==================================================

SALIDA_DAC:


;--------------------------------------------------
; Guardar copia completa
;--------------------------------------------------

mov r21, r20


;--------------------------------------------------
; Bits 0-5
;--------------------------------------------------

andi r20, 0b00111111

out PORTC, r20


;--------------------------------------------------
; Recuperar valor original
;--------------------------------------------------

mov r20, r21


;--------------------------------------------------
; Desplazar bits 6 y 7 hacia PB0-PB1
;--------------------------------------------------

lsr r20
lsr r20
lsr r20
lsr r20
lsr r20
lsr r20


andi r20, 0b00000011


;--------------------------------------------------
; Leer PORTB para conservar PB5
;--------------------------------------------------

in r21, PORTB


;--------------------------------------------------
; Borrar solamente PB0 y PB1
; PB5 NO se modifica.
;--------------------------------------------------

andi r21, 0b11111100


;--------------------------------------------------
; Agregar nuevos bits
;--------------------------------------------------

or r21, r20

out PORTB, r21


ret

;=======================================================================
; CÓDIGO PARA INTERRUPCIÓN DEL TIMER
; AUMENTAR - DISMINUIR FRECUENCIA
;=======================================================================


;-----------------------------------------------------
; Programa para detectar pulos de PD2 - PD3
;-----------------------------------------------------

ESPERAR_PULSACION:


;--------------------------------------------------
; PD2
; Disminuye tiempo
; Aumenta frecuencia
;--------------------------------------------------

sbic PIND, PD2
rjmp DISMINUIR_TIEMPO


;--------------------------------------------------
; PD3
; Aumenta tiempo
; Disminuye frecuencia
;--------------------------------------------------

sbic PIND, PD3
rjmp AUMENTAR_TIEMPO


rjmp ESPERAR_PULSACION



;==================================================
; DISMINUIR TIEMPO
;==================================================

DISMINUIR_TIEMPO:

rcall DELAY


;--------------------------------------------------
; Confirmar pulsación
;--------------------------------------------------

sbis PIND, PD2
rjmp ESPERAR_PULSACION


;--------------------------------------------------
; Dividir r17:r18 entre 2
;--------------------------------------------------

lsr r17
ror r18


;--------------------------------------------------
; Actualizar OCR1A
;--------------------------------------------------

sts OCR1AH, r17
sts OCR1AL, r18


rjmp ESPERAR_SUELTA



;==================================================
; AUMENTAR TIEMPO
;==================================================

AUMENTAR_TIEMPO:

rcall DELAY


;--------------------------------------------------
; Confirmar pulsación
;--------------------------------------------------

sbis PIND, PD3
rjmp ESPERAR_PULSACION


;--------------------------------------------------
; Multiplicar r17:r18 por 2
;--------------------------------------------------

lsl r18
rol r17


;--------------------------------------------------
; Actualizar OCR1A
;--------------------------------------------------

sts OCR1AH, r17
sts OCR1AL, r18


rjmp ESPERAR_SUELTA



;==================================================
; ESPERAR QUE BOTÓN SE SUELTE
;==================================================

ESPERAR_SUELTA:


sbic PIND, PD2
rjmp ESPERAR_SUELTA


sbic PIND, PD3
rjmp ESPERAR_SUELTA


rjmp ESPERAR_PULSACION



;===================================================
; RETARDO ANTIRREBOTE
;==================================================

DELAY:

ldi r19, 50


DELAY_1:

ldi r20, 255


DELAY_2:

dec r20
brne DELAY_2

dec r19
brne DELAY_1

ret



;==================================================
; TIMER
;==================================================

CONFIG_TIMER1:


;--------------------------------------------------
; Timer1 detenido inicialmente
;--------------------------------------------------

ldi r16, 0

sts TCCR1A, r16
sts TCCR1B, r16


;--------------------------------------------------
; Reiniciar contador
;--------------------------------------------------

ldi r16, 0

sts TCNT1H, r16
sts TCNT1L, r16


;--------------------------------------------------
; OCR1A = r17:r18
;--------------------------------------------------

sts OCR1AH, r17
sts OCR1AL, r18


;--------------------------------------------------
; Limpiar bandera OCF1A
;--------------------------------------------------

ldi r16, (1 << OCF1A)
sts TIFR1, r16


;--------------------------------------------------
; Habilitar interrupción Compare A
;--------------------------------------------------

ldi r16, (1 << OCIE1A)
sts TIMSK1, r16


;--------------------------------------------------
; CTC
; Prescaler = 1024
;--------------------------------------------------

ldi r16, (1 << WGM12) | (1 << CS12) | (1 << CS10)

sts TCCR1B, r16


;--------------------------------------------------
; Habilitar interrupciones globales
;--------------------------------------------------

sei

ret



;==================================================
; INTERRUPCIÓN TIMER1
;
; En cada interrupción:
;
; 1. Cambiar PB5
; 2. Revisar señal seleccionada
; 3. Leer su LUT
; 4. Mandar dato al DAC
; 5. Incrementar índice
;==================================================

TIMER1_COMPA:


;--------------------------------------------------
; Guardar r0
;--------------------------------------------------

push r0


;--------------------------------------------------
; Guardar SREG
;--------------------------------------------------

in r0, SREG
push r0


;--------------------------------------------------
; Guardar registros utilizados
;--------------------------------------------------

push r19
push r20
push r21
push r28
push r29


;--------------------------------------------------
; LED DE INTERRUPCIÓN
; PB5 cambia cada vez que ocurre una interrupción.
;--------------------------------------------------

sbi DDRB, DDB5
sbi PINB, PB5


;==================================================
; VER QUÉ SEÑAL FUE SELECCIONADA EN EL MENÚ
;==================================================

cpi r25, 1
breq ISR_SENAL1

cpi r25, 2
breq ISR_SENAL2


;--------------------------------------------------
; Si por alguna razón no hay señal válida
;--------------------------------------------------

rjmp ISR_FIN



;==================================================
; SEÑAL 1
;==================================================

ISR_SENAL1:


;--------------------------------------------------
; r24 = posición actual de la LUT
;--------------------------------------------------

mov r0, r24


;--------------------------------------------------
; Buscar muestra señal 1
;--------------------------------------------------

rcall get_7seg_code


;--------------------------------------------------
; Mandar muestra al DAC
;--------------------------------------------------

rcall SALIDA_DAC


;--------------------------------------------------
; Siguiente posición
;--------------------------------------------------

inc r24


rjmp ISR_FIN



;==================================================
; SEÑAL 2
;==================================================

ISR_SENAL2:


;--------------------------------------------------
; Dejamos preparado el índice.
;--------------------------------------------------

mov r0, r24


;--------------------------------------------------
; Buscar muestra señal 2
;--------------------------------------------------

rcall get_senal2_code


;--------------------------------------------------
; Mandar muestra al DAC
;--------------------------------------------------

rcall SALIDA_DAC


;--------------------------------------------------
; Avanzar posición igualmente.
;--------------------------------------------------

inc r24


;==================================================
; FIN DE INTERRUPCIÓN
;==================================================

ISR_FIN:


;--------------------------------------------------
; Restaurar registros
;--------------------------------------------------

pop r29
pop r28
pop r21
pop r20
pop r19


;--------------------------------------------------
; Restaurar SREG
;--------------------------------------------------

pop r0
out SREG, r0


;--------------------------------------------------
; Restaurar r0
;--------------------------------------------------

pop r0


reti
